[Flags]
public enum CfUpdateFlags : uint

Fields

AllowPartial = 1024

Clears the always-full state so the file can be dehydrated.

AlwaysFull = 512

Marks the file as always fully present.

ClearInSync = 64

Marks the placeholder as not in sync after a successful update.

Dehydrate = 4

Dehydrates the complete file after updating it.

DisableOnDemandPopulation = 16

Disables on-demand population for a directory.

EnableOnDemandPopulation = 8

Enables on-demand population for a directory.

MarkInSync = 2

Marks the placeholder as in sync after a successful update.

None = 0

Uses the default update behavior.

PassthroughFsMetadata = 256

Passes zero-valued metadata fields through instead of treating them as unchanged.

RemoveFileIdentity = 32

Removes the placeholder file identity.

RemoveProperty = 128

Removes all extrinsic placeholder properties.

VerifyInSync = 1

Fails the update unless the placeholder is currently in sync.
